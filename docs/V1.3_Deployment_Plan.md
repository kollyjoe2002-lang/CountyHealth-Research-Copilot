```bash

    required = {'ihme.burden', 'places.estimates', 'places.measures',
                'places.releases', 'places.measure_versions', 'geography.rucc_2023',
                'analytics.vw_current_county_lookup', 'analytics.dim_cause',
                'analytics.dim_sex', 'analytics.dim_age_group', 'analytics.dim_race'}

        'SELECT table_schema,table_name FROM information_schema.tables').fetchall()}

    assert c.execute('SELECT COUNT(*) FROM places.estimates').fetchone()[0] > 0
    assert not c.execute('SELECT 1 FROM geography.rucc_2023 WHERE source_year IS DISTINCT FROM 2023 LIMIT 1').fetchone()
```

## Freeze, backup and exact fast-forward

Before entering this block, obtain the explicit deployment go/no-go approval.
Independently confirm import writers are stopped. Operational stores are separate
files; do not overwrite their new records during a code rollback.

```bash
PREV_COMMIT="$(git rev-parse HEAD)"
git fetch origin
# Verify branch identity and the exact approved candidate; do not follow a moving
# branch head blindly, merge divergent history, reset, rebase or squash.
git cat-file -e "$RC_COMMIT^{commit}"
git merge-base --is-ancestor "$PREV_COMMIT" "$RC_COMMIT"

STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_DIR="$BACKUP_ROOT/v1.3-$STAMP"
mkdir -m 700 "$BACKUP_DIR"
printf '%s\n' "$PREV_COMMIT" > "$BACKUP_DIR/previous-commit.txt"
printf '%s\n' "$RC_COMMIT" > "$BACKUP_DIR/candidate-commit.txt"
sudo systemctl stop "$SERVICE"
# A WAL requires a separately reviewed consistent-snapshot procedure. Do not
# delete it, issue CHECKPOINT, or copy an inconsistent warehouse by guessing.
test ! -e "$DB_PATH.wal"
for file in countyhealth.duckdb epicounty_telemetry.duckdb epicounty_feedback.duckdb; do
  source="$PROD_CHECKOUT/database/$file"
  if test -f "$source"; then
    test ! -e "$source.wal"
    cp --reflink=auto --preserve=mode,timestamps -- "$source" "$BACKUP_DIR/$file"
    test "$(sha256sum "$source" | cut -d' ' -f1)" = \
         "$(sha256sum "$BACKUP_DIR/$file" | cut -d' ' -f1)"
  fi
done
sha256sum "$BACKUP_DIR"/*.duckdb > "$BACKUP_DIR/SHA256SUMS"
(cd "$BACKUP_DIR" && sha256sum -c SHA256SUMS)
# Verify the warehouse backup opens read-only as well; establish an independently
# stored/restorable copy under the approved backup policy before proceeding.
BACKUP_DB="$BACKUP_DIR/countyhealth.duckdb" "$PROD_PYTHON" - <<'PY'

    assert c.execute('SELECT COUNT(*) FROM ihme.burden').fetchone()[0] > 0

PY
# STOP unless backup restoration/independent-storage verification is approved.
```
